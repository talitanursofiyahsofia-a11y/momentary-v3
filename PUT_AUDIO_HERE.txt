<script>
const fileInput=document.getElementById("file");
const player=document.getElementById("player");
const result=document.getElementById("result");
const status=document.getElementById("status");
let recognition=null, running=false;

fileInput.addEventListener("change",()=>{
  const file=fileInput.files[0];
  if(!file)return;
  player.src=URL.createObjectURL(file);
  status.textContent="File siap diputar.";
});

const SR=window.SpeechRecognition||window.webkitSpeechRecognition;

if(SR){
  recognition=new SR();
  recognition.lang="id-ID";
  recognition.continuous=true;
  recognition.interimResults=true;

  recognition.onresult=e=>{
    let text="";
    for(let i=0;i<e.results.length;i++){
      text+=e.results[i][0].transcript+" ";
    }
    result.value=text.trim();
  };

  recognition.onerror=e=>{
    status.textContent="Transkripsi gagal: "+e.error+
      ". Untuk audio yang sudah tersimpan, browser tidak selalu bisa mengirim suara file langsung ke Speech Recognition.";
  };

  recognition.onend=()=>{
    if(running) status.textContent="Transkripsi berhenti.";
  };
}else{
  status.textContent="Speech Recognition tidak didukung browser ini.";
}

document.getElementById("start").onclick=()=>{
  if(!fileInput.files[0]){
    status.textContent="Pilih file audio terlebih dahulu.";
    return;
  }
  if(!recognition){
    status.textContent="Gunakan Chrome/Edge yang mendukung Speech Recognition.";
    return;
  }
  running=true;
  status.textContent="Transkripsi dimulai. Putar audio lalu izinkan mikrofon jika diminta.";
  recognition.start();
  player.play().catch(()=>{});
};

document.getElementById("stop").onclick=()=>{
  running=false;
  if(recognition) try{recognition.stop()}catch(e){}
  player.pause();
  status.textContent="Berhenti.";
};

document.getElementById("copy").onclick=async()=>{
  await navigator.clipboard.writeText(result.value);
  status.textContent="Hasil berhasil disalin.";
};
</script>
